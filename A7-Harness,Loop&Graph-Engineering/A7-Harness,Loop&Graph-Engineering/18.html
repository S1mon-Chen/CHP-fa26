config = {"configurable": {"thread_id": "ticket_8902"}}
audit_logs = []

for

ckpt_id = state.config["configurable"]["checkpoint_id"]
audit_logs.append({
"step": state.next,
"checkpoint_id": ckpt_id,
"state_values": state.values,
"author": state.metadata.get("created_by", "system")
})
