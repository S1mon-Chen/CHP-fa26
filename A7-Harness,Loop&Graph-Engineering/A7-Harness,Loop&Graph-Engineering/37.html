# 1. 在 E2B 沙箱中物理挂载 Patch 并运行 Pytest
def e2b_tester_node(state: RefactorSwarmState) -> dict:
with Sandbox() as sbx:
sbx.files.write("/home/user/app.py", state["patch_code"])
res = sbx.commands.run("pytest /home/user/tests")
if
res.exit_code == 0: return {"test_status": "PASSED"}
return {"test_status": "FAILED", "traceback": res.stderr}

# 2. 连续报错时触发时光旅行 (Time-Travel Rollback)
def time_travel_rollback_node(state: RefactorSwarmState) -> dict:
history = list(app.get_state_history(config))
# 提取第 1 步安全的 Triage Checkpoint ID 物理原位恢复
safe_ckpt = history[-1].config["configurable"]["checkpoint_id"]
return {"rollback_ckpt_id": safe_ckpt, "action": "RE-PLAN"}
  • 崩溃隔离（Crash Isolation）：沙箱失败不会摧毁宿主主图进程，Traceback 作为纯 Observation 灌入下一次推演。
  • 零污染物理重置：通过 `get_state_history()` 重置，彻底抹除脏代码污染。

将物理隔离沙箱 (E2B) 的运行反馈与 LangGraph Checkpointer 时光旅行机制完美缝合，是构建高韧性重构 Agent 的终极工程落地！当沙箱单测发生死锁或持续报错时，系统能够通过 get_state_history() 物理擦除不良推演路径，回溯至已知安全的快照重新发起决议，实现绝对可靠的全自动代码重构闭环。
