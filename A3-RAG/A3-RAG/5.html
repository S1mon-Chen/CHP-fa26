import re

def clean_text(raw_text: str) -> str:
    # 替换连续的空格和换行符，保留单换行
    text = re.sub(r' +', ' ', raw_text)
    text = re.sub(r'\n+', '\n', text)
    # 过滤非打印字符，保留关键制表符与换行
    text = "".join(
        ch for ch in text 
        if ch.isprintable() or ch in "\n\t"
    )
    return text.strip()
工程心得： RAG 系统的召回质量硬上限由数据清洗决定。在文本向量化（Embedding）之前，使用 isprintable() 剔除不可打印的噪点字元，是消除高维向量空间噪点干扰、提升语义检索精确度最核心的基础设施。
