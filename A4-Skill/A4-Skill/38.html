class SqlPatchAgentController:
    def __init__(self, llm_client, registry: AgentToolchainRegistry, max_steps=5):
        self.llm = llm_client
        self.registry = registry
        self.max_steps = max_steps

    def run_audit_pipeline(self, user_query: str) -> str:
        messages = [{"role": "user", "content": user_query}]
        tools_schema = self.registry.get_all_schemas()

        for step in range(self.max_steps):
            response = self.llm.chat_completion(messages, tools=tools_schema)
            messages.append(response.as_message())  # 1. 记录 Assistant 消息 ★

            if not response.tool_calls:
                return response.content  # 任务收敛完成

            for call in response.tool_calls:
                try:
                    # 2. 物理分发与副作用拦截（HITL / 沙箱）★
                    obs = self.dispatch_tool_with_governance(call)
                except Exception as err:
                    # 3. 错误 Traceback 转化为 Observation 促进自愈 ★
                    obs = f"Observation: Sandbox Exec Error: {str(err)}"

                messages.append({"role": "tool", "tool_call_id": call.id, "content": obs})

        raise MaxStepsExceededError("Agent reached maximum execution turns.")

    def dispatch_tool_with_governance(self, call) -> str:
        tool_meta = self.registry.get_tool(call.function.name)
        if tool_meta.side_effect == "STATE_CHANGING" and tool_meta.requires_approval:
            # HITL 卡扣：暂停并提示人工确认
            if not human_approval_dialog(call.function.arguments):
                return "Observation: User rejected Git patch application."

        # 进入 Docker 隔离沙箱执行，并应用 LogPruner 剪枝
        raw_res = docker_sandbox.execute(tool_meta.func, call.function.arguments)
        return LogPrunerEngine.prune(raw_res, max_lines=60)

案例控制器的四大子系统架构

4 Core Agent Subsystems

Subsystems
1. 契约分发与类型反序列化 Pydantic Dispatch

上游模型通过 OpenAPI JSON Schema 约束生成入参，宿主通过 Pydantic 动态反序列化，防范类型不匹配。

2. HITL 高危动作拦截 Approval Gate

对 `git_apply_patch_tool` 等状态修改类工具自动挂起，弹窗确认 Diff 无误后再释放进入物理写路径。

3. Docker 沙箱与超时强杀 Sandbox Exec

`sandbox_pytest_tool` 在禁网（`network_mode="none"`）、配额限制的 Docker 容器中执行单测，超时自动 `container.kill()`。

4. LogPruner 剪枝与 Self-Correction Error As Feedback

将过长单测日志压缩截断后包裹为 `role: "tool"` Observation，若有 Traceback 则引导 LLM 在 2nd-turn 自动补修。
