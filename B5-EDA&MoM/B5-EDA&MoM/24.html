2. 生产级 `OrderProducerService` 与回调函数实现 Java 17 & Spring Boot 3
// A. 生产者服务类 (`OrderProducerService.java`)
@Service
public class OrderProducerService {
@Autowired
private KafkaTemplate<String, OrderEvent> kafkaTemplate;
public void sendOrderEvent(OrderEvent event) {
// 非阻塞发送返回 CompletableFuture
CompletableFuture<SendResult<String, OrderEvent>> future =
kafkaTemplate.send("order-events", event.getOrderId(), event);
// 注册非阻塞完成回调 (Success / Exception)
future.whenComplete((result, ex) -> {
if (ex == null) {
RecordMetadata meta = result.getRecordMetadata();
log.info("Sent order [{}] to partition [{}] offset [{}]",
event.getOrderId(), meta.partition(), meta.offset());
} else {
log.error("Failed to send order [{}]", event.getOrderId(), ex);
}
});
}
}
