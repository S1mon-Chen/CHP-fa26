import json, re, subprocess
from pydantic import BaseModel, Field

# 1. Tool Call Schema Definition
class ToolCall(BaseModel):
tool_name: str = Field(description="Tool name: 'read_file','apply_patch','run_pytest'")
parameters: dict = Field(description="Key-value pairs matching tool signature")

# 2. Production System Prompt Construction
SYSTEM_PROMPT = """<system_instructions>
<role>You are a Mini-Coding-Assistant. Solve tasks via Thought-Action-Observation loop.</role>
<tools>read_file(path), apply_patch(path, patch), run_pytest(test_path)</tools>
<rules>Emit ONE action inside <action>{"tool_name": "...", "parameters": {...}}</action></rules>
</system_instructions>"""

# 3. Autonomous ReAct Agent Loop Engine
class MiniCodingAssistant:
def __init__(self, llm_client, max_steps=10):
self.llm = llm_client
self.max_steps = max_steps
self.history = [{"role": "system", "content": SYSTEM_PROMPT}]

def run_task(self, user_query):
self.history.append({"role": "user", "content": f"<task>{user_query}</task>"})
for step in range(self.max_steps):
resp = self.llm.generate(self.history)
self.history.append({"role": "assistant", "content": resp})

# Extract & Parse Tool Call
match = re.search(r"<action>(.*?)</action>", resp, re.DOTALL)
if not match:
if "<final_answer>" in resp: return "Task Completed Success"
obs = "Observation: ValidationError - Missing <action> tag."
else:
try:
action_data = json.loads(match.group(1))
tool_call = ToolCall(**action_data)
obs = self.execute_tool(tool_call)
except Exception as e:
obs = f"Observation: Syntax/Schema Error: {str(e)}"

self.history.append({"role": "user", "content": f"<observation>{obs}</observation>"})
return "Max Steps Exceeded"

def execute_tool(self, tool_call):
# Dispatch to real OS sandbox (e.g. run_pytest via subprocess)
if tool_call.tool_name == "run_pytest":
res = subprocess.run(["pytest", tool_call.parameters["path"]], capture_output=True, text=True)
return res.stdout[-1000:] # Truncate long log
# ... handle read_file, apply_patch ...
约 80 行完整可运行的代码 Agent 核心控制骨架 可直接部署运行 ★
1. 结构化 Prompt 与 Pydantic 契约 Module 1 & 2

使用 `` 隔离角色，将工具入参定义为 `ToolCall` Pydantic 模型，利用 JSON Schema 与正则结构卡死工具输出。

2. ReAct 思考循环与上下文同步 Module 3 & 4

在 `run_task` 中轮替推演 `Thought` 与 `Action`，拦截沙箱返回的 `Observation` 并追加入 `history`，保持智能体认知一致。

3. 沙箱执行与异常自愈 (Self-Healing) Robustness

当 JSON 解析失败或单测报 `FAILED` 时，不崩溃退栈，而是包装成 `Observation: [Traceback]` 投递给模型触发自动自我纠错。

01

限制最大循环步数 (`max_steps=10`)

必须在 Agent 控制循环中加入硬性计数熔断，防止出现死循环或无限重试消耗 API 预算。

02

智能截断长输出 (`res.stdout[-1000:]`)

Shell 与 Pytest 原始日志必须经过剪枝后才能喂回给 LLM，防止海量日志瞬间挤爆 Context Window。

03

从零手写掌控底层机制

抛开 LangChain/LlamaIndex 高层封装，手写这 80 行代码能够让架构师彻底吃透 Agent 的底层逻辑。

Capstone Engineering Takeaway / 案例精讲总结

写好 Mini-Coding-Assistant 的核心秘密是“用控制流把无状态 LLM 与有状态沙箱紧密缝合”；掌握这套标准模板，就能轻松扩展出企业级智能 Copilot。
