# 生产级基于 Hybrid RAG、MCP 与 Tree-sitter 的影响范围定位器
async def locate_impacted_scope(issue_body: str, vector_store, mcp_client) -> dict:
# 1. AST Chunk 粒度物理混合检索 (Dense + BM25)
retrieved_chunks = await vector_store.hybrid_search(query=issue_body, top_k=5)
primary_files = {c.metadata["file_path"] for c in retrieved_chunks}

# 2. 调用 MCP Git/LSP 服务获取跨文件依赖引用链 (Symbol Graph)
dependent_files = set()
for
fpath in primary_files:
deps = await mcp_client.call_tool("get_symbol_dependencies", {"path": fpath})
dependent_files.update(deps.get("imports", []))

# 3. 对辅助依赖文件提炼 Tree-sitter Skeleton 瘦身打包
context_map = {}
for
f in (primary_files | dependent_files):
raw_code = open(f).read()
context_map[f] = raw_code if f in primary_files else extract_skeleton(raw_code)
return {"target_files": list(primary_files), "context_xml": package_xml(context_map)}

  • 依赖链物理完整：MCP 工具精准补全漏掉的底层 `import` 类定义，防止改错重构入参。
