# 模块二 8 大子系统全栈管道串联实现
from langgraph.graph import StateGraph, END

builder = StateGraph(RefactorState)
builder.add_node("context_packager", package_context_node) # S1/S2
builder.add_node("mcp_bridge", load_mcp_tools_node) # S3
builder.add_node("coder_agent", generate_diff_patch_node) # S4
builder.add_node("diff_validator", apply_and_validate_diff_node) # S5
builder.add_node("sandbox_tester", sandboxed_testing_node) # S6
builder.add_node("hitl_approval", hitl_approval_gate_node) # S8
builder.add_node("git_transaction", commit_tx_node) # S7

# 管道控制流边与条件自愈路由定义
builder.add_edge("context_packager", "mcp_bridge")
builder.add_edge("mcp_bridge", "coder_agent")
builder.add_edge("coder_agent", "diff_validator")
builder.add_edge("diff_validator", "sandbox_tester")
builder.add_conditional_edges("sandbox_tester", route_test_feedback, {
"PASSED": "hitl_approval",
"FAILED": "coder_agent" # TDD 自愈重修图环
})
builder.add_edge("hitl_approval", "git_transaction")
builder.add_edge("git_transaction", END)
app = builder.compile(checkpointer=PostgresSaver(pool))
  • 完整状态持久化：结合 `PostgresSaver` 支持长任务崩溃续传与全流程节点可追溯。
  • 闭环 TDD 自愈与人工卡扣：只有 Pytest 全绿且通过 HITL 审批的代码才能触发原子 Commit 写盘。
物理本质：用标准状态图将 8 大子系统缝合为真正具备生产可用性的全栈 AI 系统
