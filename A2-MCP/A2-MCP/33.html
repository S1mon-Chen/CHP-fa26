Spring AI 中的资源与错误处理

利用 Resource 抽象与异常捕获机制，构建高容错的 MCP 服务

统一的 Resource 抽象

通过 Spring 核心 ResourceLoader 接口，无缝切换 classpath:、file: 或网络存储，提供一致的 I/O 体验。

高容错错误处理

必须拦截资源读取过程中的 I/O 异常。通过捕获机制返回结构化错误描述，避免进程崩溃并让 LLM 能够感知异常原因。

ResourceHandlerService.java 
import org.springframework.core.io.*;
import org.springframework.ai.mcp.annotations.McpTool;
import org.springframework.stereotype.Service;

@Service
public class ResourceHandlerService {
    private final ResourceLoader resourceLoader;
    public ResourceHandlerService(ResourceLoader loader) {
        this.resourceLoader = loader;
    }

    @McpTool(description = "Read internal docs")
    public String readDoc(String docId) {
        try {
            // 1. 统一加载资源
            Resource res = resourceLoader.getResource("file:/data/"+docId);
            if (!res.exists()) return "Error: Not found.";

            return new String(res.getInputStream().readAllBytes());
        } catch (Exception e) {
            // 2. 异常捕获，保障 Server 健壮性
            return "I/O Error: " + e.getMessage();
        }
    }
}
